SET client_min_messages = warning;
SET search_path TO public;

DROP FUNCTION IF EXISTS upsert_run_metrics CASCADE;
DROP VIEW IF EXISTS v_model_fixture_summary CASCADE;
DROP VIEW IF EXISTS v_long_context_analysis CASCADE;
DROP TABLE IF EXISTS request_metrics CASCADE;
DROP TABLE IF EXISTS runs CASCADE;
DROP TABLE IF EXISTS fixtures CASCADE;
DROP TABLE IF EXISTS models CASCADE;

CREATE TABLE models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    family VARCHAR(50) NOT NULL,
    parameter_count BIGINT CHECK (parameter_count > 0),
    context_length INT CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE fixtures (
    fixture_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    kind VARCHAR(50) NOT NULL CHECK (kind IN ('short', 'medium', 'long', 'code')),
    token_count INT CHECK (token_count >= 0),
    description TEXT
);

CREATE TABLE runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id),
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id),
    started_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    finished_at TIMESTAMP,
    status VARCHAR(20) NOT NULL DEFAULT 'completed' CHECK (status IN ('completed', 'failed', 'timeout')),
    config_json JSONB DEFAULT '{}'::jsonb
);

CREATE TABLE request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id),
    timestamp TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    input_tokens INT CHECK (input_tokens >= 0),
    output_tokens INT CHECK (output_tokens >= 0),
    latency_ms FLOAT CHECK (latency_ms >= 0),
    mtp_accepted BOOL DEFAULT FALSE,
    error_code VARCHAR(50)
);

CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_timestamp ON request_metrics(timestamp);
CREATE INDEX idx_models_name ON models(name);
CREATE INDEX idx_fixtures_kind ON fixtures(kind);

CREATE OR REPLACE FUNCTION upsert_run_metrics(
    p_run_id INT,
    p_input_tokens INT,
    p_output_tokens INT,
    p_latency_ms FLOAT,
    p_mtp_accepted BOOL
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (run_id, input_tokens, output_tokens, latency_ms, mtp_accepted)
    VALUES (p_run_id, p_input_tokens, p_output_tokens, p_latency_ms, p_mtp_accepted)
    ON CONFLICT (run_id) DO UPDATE SET
        input_tokens = EXCLUDED.input_tokens,
        output_tokens = EXCLUDED.output_tokens,
        latency_ms = EXCLUDED.latency_ms,
        mtp_accepted = EXCLUDED.mtp_accepted;
END;
$$ LANGUAGE plpgsql;
