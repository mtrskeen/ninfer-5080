DROP TABLE IF EXISTS request_metrics CASCADE;
DROP TABLE IF EXISTS runs CASCADE;
DROP TABLE IF EXISTS fixtures CASCADE;
DROP TABLE IF EXISTS models CASCADE;

CREATE TABLE models (
    model_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL UNIQUE,
    architecture VARCHAR(50) NOT NULL CHECK (architecture IN ('transformer', 'mamba', 'hybrid')),
    context_length INT NOT NULL CHECK (context_length > 0),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE fixtures (
    fixture_id SERIAL PRIMARY KEY,
    kind VARCHAR(50) NOT NULL CHECK (kind IN ('short_qa', 'long_doc', 'code_gen', 'multi_turn')),
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE runs (
    run_id SERIAL PRIMARY KEY,
    model_id INT NOT NULL REFERENCES models(model_id) ON DELETE CASCADE,
    fixture_id INT NOT NULL REFERENCES fixtures(fixture_id) ON DELETE CASCADE,
    started_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    finished_at TIMESTAMP,
    status VARCHAR(20) NOT NULL DEFAULT 'completed' CHECK (status IN ('completed', 'failed', 'timeout')),
    UNIQUE(model_id, fixture_id, started_at)
);

CREATE TABLE request_metrics (
    metric_id SERIAL PRIMARY KEY,
    run_id INT NOT NULL REFERENCES runs(run_id) ON DELETE CASCADE,
    input_tokens INT NOT NULL CHECK (input_tokens >= 0),
    output_tokens INT NOT NULL CHECK (output_tokens >= 0),
    latency_ms FLOAT NOT NULL CHECK (latency_ms > 0),
    mtp_acceptance_rate FLOAT CHECK (mtp_acceptance_rate BETWEEN 0.0 AND 1.0),
    memory_peak_mb FLOAT CHECK (memory_peak_mb > 0),
    timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_models_name ON models(name);
CREATE INDEX idx_fixtures_kind ON fixtures(kind);
CREATE INDEX idx_runs_model ON runs(model_id);
CREATE INDEX idx_runs_fixture ON runs(fixture_id);
CREATE INDEX idx_runs_status ON runs(status);
CREATE INDEX idx_metrics_run ON request_metrics(run_id);
CREATE INDEX idx_metrics_latency ON request_metrics(latency_ms);
CREATE INDEX idx_metrics_mtp ON request_metrics(mtp_acceptance_rate);

CREATE VIEW v_throughput AS
SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    COUNT(rm.metric_id) AS total_requests,
    SUM(rm.output_tokens)::FLOAT / SUM(rm.latency_ms / 1000.0) AS tokens_per_second,
    AVG(rm.latency_ms) AS avg_latency_ms
FROM request_metrics rm
JOIN runs r ON rm.run_id = r.run_id
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
WHERE r.status = 'completed'
GROUP BY m.name, f.kind;

CREATE VIEW v_variability AS
SELECT 
    m.name AS model_name,
    f.kind AS fixture_kind,
    STDDEV(rm.latency_ms) AS latency_stddev,
    COALESCE(STDDEV(rm.latency_ms) / NULLIF(AVG(rm.latency_ms), 0), 0) AS cv_ratio,
    PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY rm.latency_ms) AS p95_latency_ms
FROM request_metrics rm
JOIN runs r ON rm.run_id = r.run_id
JOIN models m ON r.model_id = m.model_id
JOIN fixtures f ON r.fixture_id = f.fixture_id
WHERE r.status = 'completed'
GROUP BY m.name, f.kind;

CREATE OR REPLACE FUNCTION upsert_request_metric(
    p_run_id INT,
    p_input_tokens INT,
    p_output_tokens INT,
    p_latency_ms FLOAT,
    p_mtp_acceptance FLOAT,
    p_memory_peak FLOAT
) RETURNS VOID AS $$
BEGIN
    INSERT INTO request_metrics (run_id, input_tokens, output_tokens, latency_ms, mtp_acceptance_rate, memory_peak_mb)
    VALUES (p_run_id, p_input_tokens, p_output_tokens, p_latency_ms, p_mtp_acceptance, p_memory_peak)
    ON CONFLICT (run_id, input_tokens, output_tokens, latency_ms) DO NOTHING;
END;
$$ LANGUAGE plpgsql;
