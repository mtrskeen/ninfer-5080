## Request Metrics Service

A complete, self-contained TypeScript 5 service. It parses NDJSON incrementally (bounded memory), validates records with hand-rolled checks (no schema libs), groups by `model`/`fixture`, computes mean / median / sample standard deviation, and exposes `/health`, `/summary`, and `/runs/:id`.

### `src/statistics.ts`

Pure math. Uses Welford's online algorithm for numerically stable mean/sample variance, and a rank-based median (no sorting, O(n)).

```ts
/**
 * Online accumulator for mean and sample variance via Welford's algorithm.
 * Numerically stable for large N without catastrophic cancellation.
 */
export class RunningStats {
  private n = 0;
  private mean = 0;
  private m2 = 0;
  private min = Number.POSITIVE_INFINITY;
  private max = Number.NEGATIVE_INFINITY;
  private values: number[] = [];

  add(x: number): void {
    if (!Number.isFinite(x)) throw new TypeError('RunningStats.add requires a finite number');
    this.n += 1;
    const delta = x - this.mean;
    this.mean += delta / this.n;
    this.m2 += delta * (x - this.mean);
    if (x < this.min) this.min = x;
    if (x > this.max) this.max = x;
    this.values.push(x);
  }

  get count(): number { return this.n; }
  get average(): number { return this.n === 0 ? NaN : this.mean; }
  get minimum(): number { return this.n === 0 ? NaN : this.min; }
  get maximum(): number { return this.n === 0 ? NaN : this.max; }

  /** Sample standard deviation (n-1 denominator). Null when fewer than 2 samples. */
  sampleStdDev(): number | null {
    if (this.n < 2) return null;
    return Math.sqrt(this.m2 / (this.n - 1));
  }

  /** Median via quickselect on the retained samples. Empty -> null. */
  median(): number | null {
    if (this.n === 0) return null;
    const arr = this.values.slice();
    const target = (this.n + 1) >> 1; // 1-based rank of the lower-middle element
    const value = quickselect(arr, target);
    return value ?? null;
  }

  /** Drop the oldest samples beyond `cap` (FIFO) to bound memory. */
  trimTo(cap: number): void {
    if (this.values.length <= cap) return;
    const drop = this.values.length - cap;
    this.values.splice(0, drop);
    this.recompute();
  }

  /** Re-derive moments from the surviving samples after trimming. */
  private recompute(): void {
    let sum = 0;
    let mn = Number.POSITIVE_INFINITY;
    let mx = Number.NEGATIVE_INFINITY;
    for (const v of this.values) {
      sum += v;
      if (v < mn) mn = v;
      if (v > mx) mx = v;
    }
    this.n = this.values.length;
    this.min = this.n === 0 ? Number.POSITIVE_INFINITY : mn;
    this.max = this.n === 0 ? Number.NEGATIVE_INFINITY : mx;
    if (this.n === 0) { this.mean = 0; this.m2 = 0; return; }
    this.mean = sum / this.n;
    let sse = 0;
    for (const v of this.values) { const d = v - this.mean; sse += d * d; }
    this.m2 = sse;
  }
}

/**
 * In-place selection of the k-th smallest element (1-based).
 * Returns undefined if k is out of range.
 */
function quickselect(a: number[], k: number): number | undefined {
  if (k < 1 || k > a.length) return undefined;
  let lo = 0;
  let hi = a.length - 1;
  while (lo < hi) {
    const pivot = a[(lo + hi) >>> 1]!;
    let i = lo;
    let j = hi;
    while (i <= j) {
      while (a[i]! < pivot) i++;
      while (a[j]! > pivot) j--;
      if (i <= j) { const t = a[i]!; a[i] = a[j]!; a[j] = t; i++; j--; }
    }
    if (k <= j) hi = j;
    else if (k > i) lo = i;
    else break;
  }
  return a[k - 1];
}
```

### `src/validator.ts`

Hand-rolled validation — no schema library. Produces precise, field-level issues.

```ts
import type { RequestMetricRecord, ValidationIssue, ValidationErrorCode } from './types.js';

interface FieldSpec {
  key: keyof RequestMetricRecord;
  required: boolean;
  type: 'string' | 'number' | 'object';
  numericRule?: (v: number) => string | null;
  stringRule?: (v: string) => string | null;
}

const NON_EMPTY_STRING = (v: string) => (v.trim().length === 0 ? 'must be non-empty' : null);
const POSITIVE_FINITE = (v: number) =>
  !Number.isFinite(v) ? 'must be a finite number' : v < 0 ? 'must be >= 0' : null;

const SPEC: FieldSpec[] = [
  { key: 'runId', required: true, type: 'string', stringRule: NON_EMPTY_STRING },
  { key: 'model', required: true, type: 'string', stringRule: NON_EMPTY_STRING },
  { key: 'fixture', required: true, type: 'string', stringRule: NON_EMPTY_STRING },
  { key: 'timestamp', required: true, type: 'string', stringRule: validateTimestamp },
  { key: 'latencyMs', required: true, type: 'number', numericRule: POSITIVE_FINITE },
  { key: 'throughputTokensPerSec', required: true, type: 'number', numericRule: POSITIVE_FINITE },
  { key: 'tags', required: false, type: 'object' },
];

const KNOWN_KEYS: ReadonlySet<string> = new Set(SPEC.map((s) => s.key as string));

function validateTimestamp(v: string): string | null {
  if (v.trim().length === 0) return 'must be non-empty';
  const ms = Date.parse(v);
  if (Number.isNaN(ms)) return 'must be a valid ISO-8601 timestamp';
  return null;
}

function typeLabel(t: FieldSpec['type']): string {
  return t === 'string' ? 'a string' : t === 'number' ? 'a number' : 'a plain object';
}

function describeType(value: unknown): string {
  if (value === null) return 'null';
  if (Array.isArray(value)) return 'an array';
  return typeof value;
}

/**
 * Validate a parsed JSON value into a strict record, collecting every issue.
 * Never throws; returns `{ record, issues }`.
 */
export function validateRecord(
  value: unknown,
  line: number,
  opts: { allowUnknownFields?: boolean } = {},
): { record: RequestMetricRecord | null; issues: ValidationIssue[] } {
  const allowUnknown = opts.allowUnknownFields ?? false;
  const issues: ValidationIssue[] = [];

  if (typeof value !== 'object' || value === null || Array.isArray(value)) {
    issues.push({
      line,
      code: 'NOT_OBJECT',
      field: '$',
      message: `record must be a JSON object, got ${describeType(value)}`,
    });
    return { record: null, issues };
  }

  const obj = value as Record<string, unknown>;

  for (const spec of SPEC) {
    const present = spec.key in obj;
    if (!present && spec.required) {
      issues.push({ line, code: 'MISSING_FIELD', field: spec.key, message: `missing required field "${spec.key}"` });
      continue;
    }
    if (!present) continue;

    const v = obj[spec.key]!;

      if (typeof v !== 'string') {
        issues.push({ line, code: 'WRONG_TYPE', field: spec.key, message: `"${spec.key}" must be ${typeLabel(spec.type)}, got ${describeType(v)}` });
        continue;
      }
      const err = spec.stringRule?.(v);
      if (err) issues.push({ line,
