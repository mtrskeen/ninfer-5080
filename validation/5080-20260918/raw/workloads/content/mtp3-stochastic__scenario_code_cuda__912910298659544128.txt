# CUDA 13.1 / C++20 — Row-wise BF16 Normalization on sm_120a

Below is a complete, self-contained implementation. It uses the **CUDA 13.1** runtime API (including `cudaGetDriverEntryPoint` for device-side PTX generation) and the **C++20** standard (`std::span`, `std::bit_cast`, concepts, ranges). The program:

1. Parses command-line arguments.
2. Generates deterministic input (seeded PRNG → BF16 values).
3. Allocates RAII-managed device buffers.
4. Runs a vectorized CUDA kernel that normalizes each row to unit L2 norm in BF16.
5. Computes an FP32 reference on the host (numerically stable two-pass algorithm).
6. Performs warm-up, then timed runs using CUDA events.
7. Verifies results against the reference within a tolerance.
8. Reports timing and verification status.

// Deterministic PRNG (xorshift128+) – fast, reproducible across platforms
// ---------------------------------------------------------------------------
struct XorShift128Plus {
    uint64_t s0;
    uint64_t s1;

    explicit XorShift128Plus(uint64_t seed) : s0(seed), s1(seed ^ 0x9E3779B97F4A7C15ULL) {}

    [[nodiscard]] uint64_t next() noexcept {
        const uint64_t x = s0;
        const uint64_t y = s1;
        s0 = y;
        s1 = x ^ (x << 11) ^ (y ^ (y >> 30));
        return s1 + y;
    }

    /// Uniform float in [0, 1)
    [[nodiscard]] float nextFloat() noexcept {
        return static_cast<float>(next()) * (1.0f / 4294967296.0f);
    }

    /// Standard-normal via Box-Muller (one call produces one sample)
    [[nodiscard]] float nextNormal() noexcept {
        // We generate two uniforms per call to keep it simple; the second is
        // cached implicitly by re-deriving. For determinism we just use one
        // uniform mapped through inverse-CDF approximation (Abramowitz & Stegun).
        float u = nextFloat();
        // Clamp away from 0 and 1 for log
        if (u < 1e-7f) u = 1e-7f;
        if (u > 1.0f - 1e-7f) u = 1.0f - 1e-7f;
        // Ziggurat-free approximation: use erf inverse approx (fast enough)
        // Simpler: use the fact that for our purposes any reasonable spread works.
        // Use tan-based inverse CDF for speed:
        float z = std::sqrt(-2.0f * std::log(u)) * std::cos(2.0f * M_PI * nextFloat());
        return z;
    }
};

// RAII wrapper for pinned/device memory

template <typename T>
class DeviceBuffer {
public:
    DeviceBuffer() = default;

    explicit DeviceBuffer(size_t count) : m_count(count) {
        if (m_count == 0) return;
        CUDA_CHECK(cudaMalloc(&m_ptr, m_count * sizeof(T)));
        m_allocated = true;
    }

    ~DeviceBuffer() {
        if (m_allocated && m_ptr) {
            cudaFree(m_ptr);
        }
    }

    // Non-copyable
    DeviceBuffer(const DeviceBuffer&) = delete;
    DeviceBuffer& operator=(const DeviceBuffer&) = delete;

    // Movable
    DeviceBuffer(DeviceBuffer&& other) noexcept
        : m_ptr(other.m_ptr), m_count(other.m_count), m_allocated(other.m_allocated) {
        other.m_ptr = nullptr;
        other.m_count = 0;
        other.m_allocated = false;
    }

    DeviceBuffer& operator=(DeviceBuffer&& other) noexcept {
        if (this != &other) {
            reset();
            m_ptr = other.m_ptr;
            m_count = other.m_count;
            m_allocated = other.m_allocated;
            other.m_ptr = nullptr;
            other.m_count = 0;
            other.m_allocated = false;
        }
        return *this;
    }

    void reset() {
        if (m_allocated && m_ptr) {
            cudaFree(m_ptr);
        }
        m_ptr = nullptr;
        m_count = 0;
        m_allocated = false;
    }

    [[nodiscard]] T* data() noexcept { return m_ptr; }
    [[nodiscard]] const T* data() const noexcept { return m_ptr; }
    [[nodiscard]] size_t count() const noexcept { return m_count; }
    [[nodiscard]] bool valid() const noexcept { return m_allocated && m_ptr != nullptr; }

private:
    T* m_ptr = nullptr;
    size_t m_count = 0;
    bool m_allocated = false;
};

[[nodiscard]] inline float bf16ToFloat(__nv_bfloat16 h) noexcept {
    return __bfloat162float(h);
}

[[nodiscard]] inline __nv_bfloat16 floatToBf16(float f) noexcept {
    return __float2bfloat16(f);
}

// Convert packed __nv_bfloat162 to two floats
[[nodiscard]] inline void bf162ToFloats(__nv_bfloat162 h2, float& lo, float& hi) noexcept {
    const auto* p = reinterpret_cast<const __nv_bfloat16*>(&h2);
    lo = __bfloat162float(p[0]);
    hi = __bfloat162float(p[1]);
}

// ---------------------------------------------------------------------------
// Vectorized CUDA kernel: row-wise L2 normalization in BF16
//
// Memory layout:
//   Input:  A[rows][cols] stored as a flat array of __nv_bfloat16 (row-major).
//           Each row has `cols` elements. We process 8 elements (4 x bf162)
//           per thread per iteration for coalescing.
//
// Launch geometry:
//   Grid:  (ceil(cols / VEC_WIDTH), rows)  where VEC_WIDTH = 8
//   Block: 256 threads
//
// Algorithm (two-phase within kernel):
//   Phase 1: Each block processes one or more full rows. Threads cooperatively
//            compute partial sums of squares (in FP32) and reduce to a single
//            norm per row using shared-memory tree reduction.
//   Phase 2: After grid-wide sync (via global atomic or cooperative groups),
//            each thread divides its assigned elements by the row norm and
//            writes back BF16.
//
// For simplicity and correctness we use a simpler approach:
//   One block handles exactly one row (requires cols <= blockDim.x * UNROLL).
//   If cols > blockDim.x * UNROLL, multiple blocks cooperate on one row via
//   atomics, but for typical sizes (cols=8192, block=256, unroll=8 => 2048
//   per pass) we need ceil(8192/2048)=4 passes. We handle this by having
//   gridDim.y = rows and gridDim.x = numBlocksPerRow, with inter-block
//   communication via global memory for the norm.
//
// To keep the kernel self-contained and avoid cooperative launch complexity,
// we use a two-kernel strategy:
//   Kernel 1: Compute per-row norms (L2) into a float array.
//   Kernel 2: Normalize each element by dividing by the precomputed norm.
//
// This is numerically equivalent to a single-pass approach and avoids
// needing cooperative groups or multi-pass synchronization within one kernel.
// ---------------------------------------------------------------------------

constexpr int VEC_WIDTH = 8;       // 4 x __nv_bfloat162
constexpr int BLOCK_SIZE = 256;

// ---------------------------------------------------------------------------
// Kernel 1: Compute per-row L2 norms
//
// Grid:  (numBlocksPerRow, rows)
// Block: BLOCK_SIZE
//
// Each block reduces a chunk of the row to a partial sum-of-squares, then
// atomically adds to the global norm accumulator for that row.
// ---------------------------------------------------------------------------
__global__ void computeRowNormsKernel(
    const __nv_bfloat16* __restrict__ input,
    float* __restrict__ rowNorms,
    int rows,
    int cols,
    int numBlocksPerRow)
{
    const int row = blockIdx.y;
    const int colOffset = blockIdx.x * (BLOCK_SIZE * VEC_WIDTH);

    // Base pointer for this thread's slice in this row
    const size_t baseIdx = static_cast<size_t>(row) * cols + colOffset;

    // Local accumulation in FP32
    float localSum = 0.0f;

    // Process up to VEC_WIDTH * BLOCK_SIZE elements starting at colOffset
    // Each thread handles VEC_WIDTH consecutive elements
    const int tid = threadIdx.x;
    const int localCol = colOffset + tid * VEC_WIDTH;

    // Load 4 x bf162 (8 bf16) if within bounds
    // We guard against out-of-bounds access
    const int remaining = cols - colOffset;

    for (int i = 0; i < VEC_WIDTH; ++i) {
        const int c = localCol + i;
        if (c >= cols) break;
        const float val = bf16ToFloat(input[baseIdx + i]);
        localSum += val * val;
    }

    // Shared-memory tree reduction
    __shared__ float sdata[BLOCK_SIZE];
    sdata[tid] = localSum;
    __syncthreads();

    // Tree reduction
    for (int stride = BLOCK_SIZE / 2; stride > 0; stride >>= 1) {
        if (tid < stride) {
            sdata[tid] += sdata[tid + stride];
        }
        __syncthreads();
    }

    // Thread 0 writes partial result to global accumulator
    if (tid == 0) {
        // Atomic add to the running sum for this row
        atomicAdd(&rowNorms[row], sdata[0]);
    }
}

// ---------------------------------------------------------------------------
// Kernel 2: Normalize elements by dividing by row norm
//
// Grid:  (numBlocksPerRow, rows)
// Block: BLOCK_SIZE
//
// Each thread loads 8 BF16 values, converts to FP32, divides by the
// precomputed row norm, converts back to BF16, and stores.
// ---------------------------------------------------------------------------
__global__ void normalizeKernel(
    const __nv_bfloat16* __restrict__ input,
    __nv_bfloat16* __restrict__ output,
    const float* __restrict__ rowNorms,
    int rows,
    int cols,
    int numBlocksPerRow)
{
    const int row = blockIdx.y;
    const int colOffset = blockIdx.x * (BLOCK_SIZE * VEC_WIDTH);

    const size_t baseIdx = static_cast<size_t>(row) * cols + colOffset;
    const float invNorm = (rowNorms[row] > 0.0f) ? (1.0f / rowNorms[row]) : 0.0f;

    const int tid = threadIdx.x;
    const int localCol = colOffset + tid * VEC_WIDTH;

    // Load 4 x bf162
    // We load as individual bf16 for clarity; a production kernel would
    // use vectorized loads (uint2 for 4 bytes = 2 bf16, etc.)
    for (int i = 0; i < VEC_WIDTH; ++i) {
        const int c = localCol + i;
        if (c >= cols) break;
        const float val = bf16ToFloat(input[baseIdx + i]);
        const float normalized = val * invNorm;
        output[baseIdx + i] = floatToBf16(normalized);
    }
}

// ---------------------------------------------------------------------------
// Host-side FP32 reference: numerically stable two-pass L2 normalization
//
// Pass 1: Compute L2 norm using pairwise summation in FP32 for stability.
// Pass 2: Divide each element by the norm.
//
// This serves as the ground truth against which the GPU BF16 result is
// compared.
// ---------------------------------------------------------------------------
struct ReferenceResult {
    std::vector<float> norms;       // Per-row L2 norms (FP32)
    std::vector<float> normalized;  // Per-element normalized values (FP32)
};

static ReferenceResult computeReferenceFP32(
    const std::vector<__nv_bfloat16>& input,
    int rows,
    int cols)
{
    ReferenceResult result;
    result.norms.resize(rows);
    result.normalized.resize(static_cast<size_t>(rows) * cols);

    for (int r = 0; r < rows; ++r) {
        const size_t rowStart = static_cast<size_t>(r) * cols;

        // Pass
