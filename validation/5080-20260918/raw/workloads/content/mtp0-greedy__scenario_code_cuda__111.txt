# CUDA 13.1 / C++20 — Row-wise BF16 Normalization on sm_120a

Below is a complete, self-contained program plus its CMake project. It:

- Parses CLI arguments (`-n`, `-k`, `-iters`, `-warmup`, `-seed`, `-tol`, `-check`, `-help`).
- Uses RAII wrappers for device memory and CUDA events.
- Generates a **deterministic** BF16 matrix (seeded LCG → FP32 → round-to-nearest-even BF16).
- Runs a **vectorized** `__nv_bfloat16` kernel that normalizes each row to unit L2 norm in a single pass, using `__nv_bfloat162` (32-bit) loads/stores and `__nv_bfloat1622` (64-bit) loads/stores for 128-bit vectorization on sm_120a.
- Computes a **numerically stable FP32 reference** (two-pass: pass 1 computes the norm in FP32, pass 2 divides).
- Performs **tolerance-based verification** (max absolute error in BF16 units, plus a relative check).
- Uses **CUDA events** for timing, with a warm-up phase excluded from the timed region.
- Includes full **error checking** via a `CUDA_CHECK` macro.

## `bf16_row_norm.cu`

// Run:    ./bf16_row_norm -n 4096 -k 4096 -iters 10 -warmup 3 -seed 42 -tol 0.01 -check

// ---------------------------------------------------------------------------
// RAII device buffer
// ---------------------------------------------------------------------------
class DeviceBuffer {
public:
    DeviceBuffer() = default;

    explicit DeviceBuffer(size_t bytes) {
        if (bytes == 0) return;
        CUDA_CHECK(cudaMalloc(&ptr_, bytes));
        size_ = bytes;
    }

    DeviceBuffer(const DeviceBuffer&) = delete;
    DeviceBuffer& operator=(const DeviceBuffer&) = delete;

    DeviceBuffer(DeviceBuffer&& other) noexcept
        : ptr_(other.ptr_), size_(other.size_) {
        other.ptr_ = nullptr;
        other.size_ = 0;
    }

    DeviceBuffer& operator=(DeviceBuffer&& other) noexcept {
        if (this != &other) {
            reset();
            ptr_ = other.ptr_;
            size_ = other.size_;
            other.ptr_ = nullptr;
            other.size_ = 0;
        }
        return *this;
    }

    ~DeviceBuffer() { reset(); }

    void reset() {
        if (ptr_) {
            cudaFree(ptr_);
            ptr_ = nullptr;
            size_ = 0;
        }
    }

    void* data() const { return ptr_; }
    size_t size() const { return size_; }
    bool valid() const { return ptr_ != nullptr; }

private:
    void* ptr_ = nullptr;
    size_t size_ = 0;
};

// ---------------------------------------------------------------------------

class EventTimer {

        CUDA_CHECK(cudaEventCreate(&start_));
        CUDA_CHECK(cudaEventCreate(&stop_));

        cudaEventDestroy(start_);

    float elapsedMs() {
        CUDA_CHECK(cudaEventSynchronize(stop_));
        float ms = 0.0f;

        return ms;
    }

private:
    cudaEvent_t start_ = nullptr;
    cudaEvent_t stop_  = nullptr;
};

// ---------------------------------------------------------------------------
// Deterministic PRNG (xorshift64* for speed, seeded from user)
// ---------------------------------------------------------------------------
class Xorshift64 {
public:
    explicit Xorshift64(uint64_t seed) : state_(seed ? seed : 0x1234567890ABCDEFull) {}

    uint64_t next() {
        uint64_t x = state_;
        x ^= x >> 12;
        x ^= x << 25;
        x ^= x >> 27;
        state_ = x;
        return x * 0x2545F4914F6CDD1DULL;
    }

    // Uniform float in [0, 1)
    float nextFloat() {
        uint64_t r = next();
        return static_cast<float>(r >> 40) * (1.0f / 16777216.0f); // 24-bit mantissa
    }

    // Standard normal via Box-Muller (pairs)
    void nextNormal(float& x, float& y) {
        float u1 = nextFloat();
        float u2 = nextFloat();
        if (u1 < 1e-12f) u1 = 1e-12f;
        float r = std::sqrt(-2.0f * std::log(u1));
        float theta = 2.0f * M_PI * u2;
        x = r * std::cos(theta);
        y = r * std::sin(theta);
    }

private:
    uint64_t state_;
};

// ---------------------------------------------------------------------------
// BF16 conversion helpers (round-to-nearest-even)
// ---------------------------------------------------------------------------
static inline uint16_t floatToBf16Bits(float f) {
    // Take the upper 16 bits of the IEEE-754 single-precision representation,
    // with round-to-nearest-even.
    uint32_t bits;
    std::memcpy(&bits, &f, sizeof(f));
    // Add rounding bias: 0x7FFF + LSB of the discarded mantissa bit
    uint32_t rounded = bits + 0x7FFFu;
    // If the addition caused a carry into the sign bit, that's fine (handles
    // rounding of the exponent). For round-to-nearest-even, if the discarded
    // bits are exactly 0x8000 (half), we round to even.
    // The simple +0x7FFF trick gives round-half-up. For true RNE:
    //   if (bits & 0xFFFF) == 0x8000, round to even.
    // We implement proper RNE:
    uint32_t discard = bits & 0xFFFFu;
    uint32_t lsb = (bits >> 16) & 1u;  // least significant bit of the kept 16-bit field
    if (discard > 0x8000u || (discard == 0x808000u && (lsb & 1u))) {
        // Wait, discard is 16 bits, max 0xFFFF. Let me redo.
    }
    // Proper RNE:
    //   keep = bits >> 16
    //   discard = bits & 0xFFFF
    //   if discard > 0x8000: round up
    //   if discard == 0x8000: round to even (if keep is odd, round up)
    //   if discard < 0x8000: truncate
    uint32_t keep = bits >> 16;
    if (discard > 0x8000u) {
        keep += 1;
    } else if (discard == 0x8000u) {
        if (keep & 1u) keep += 1;
    }
    return static_cast<uint16_t>(keep);
}

static inline float bf16BitsToFloat(uint16_t h) {
    uint32_t bits = static_cast<uint32_t>(h) << 16;
    float f;
    std::memcpy(&f, &bits, sizeof(f));
    return f;
}

// ---------------------------------------------------------------------------
// Host-side reference: generate deterministic BF16 matrix
// ---------------------------------------------------------------------------
static std::vector<uint16_t> generateBf16Matrix(size_t rows, size_t cols, uint64_t seed) {
    std::vector<uint16_t> data(rows * cols);
    Xorshift64 rng(seed);
    float a, b;
    for (size_t i = 0; i < rows * cols; i += 2) {
        rng.nextNormal(a, b);
        // Scale to a reasonable range so norms are well-conditioned
        data[i]     = floatToBf16Bits(a * 0.5f);
        if (i + 1 < rows * cols)
            data[i + 1] = floatToBf16Bits(b * 0.5f);
    }
    return data;
}

// ---------------------------------------------------------------------------
// Host-side FP32 reference: row-wise L2 normalization
// ---------------------------------------------------------------------------
static std::vector<float> referenceNormalize(const std::vector<uint16_t>& bf16Data,
                                             size_t rows, size_t cols) {
    std::vector<float> result(rows * cols);
    for (size_t r = 0; r < rows; r++) {
        const uint16_t* rowPtr = &bf16Data[r * cols];
        float* outPtr = &result[r * cols];

        // Pass 1: compute L2 norm in FP32
        float sumSq = 0.0f;
        for (size_t c = 0; c < cols; c++) {
            float v = bf16BitsToFloat(rowPtr[c]);
            sumSq += v * v;
        }
        float norm = std::sqrt(sumSq);
        float inv = (norm > 0.0f) ? (1.0f / norm) : 0.0f;

        // Pass 2: divide
        for (size_t c = 0; c < cols; c++) {
            float v = bf16BitsToFloat(rowPtr[c]);
            outPtr[c] = v * inv;
        }
    }
    return result;
}

// ---------------------------------------------------------------------------
// CUDA kernel: vectorized row-wise L2 normalization in BF16
//
// Memory layout:
//   The matrix is stored in row-major order as an array of __nv_bfloat16.
//   Each row has `cols` elements. We process each row independently.
//
// Launch geometry:
//   - One block per row (gridDim.x = rows).
//   - Within a block, threads cooperate to process the row in two passes:
//       Pass 1: each thread accumulates a partial sum of squares (in FP32)
//               over its assigned elements, then we reduce within the block.
//       Pass 2: each thread divides its elements by the block-wide norm.
//
// Vectorization:
//   - We use __nv_bfloat162 (32-bit) for 2-element loads/stores.
//   - For 128-bit (8-element) vectorization we use __nv_bfloat1622 (64-bit)
//     pairs, loading two __nv_bfloat1622 per 128-bit transaction.
//   - The kernel assumes cols is a multiple of 8 for the fast path;
//     a scalar tail handles any remainder.
// ---------------------------------------------------------------------------

// 128-bit aligned type for vectorized BF16 loads (8 BF16 elements)
struct alignas(16) Bf16x8 {
    __nv_bfloat162 v[4];
};

__global__ void rowNormalizeBf16Kernel(const __nv_bfloat16* __restrict__ input,
                                       __nv_bfloat16* __restrict__ output,
                                       int rows, int cols) {
    int row = blockIdx.x;
    if (row >= rows) return;

    const __nv_bfloat16* rowIn  = input  + static_cast<size_t>(row) * cols;
    __nv_bfloat16*       rowOut = output + static_cast<size_t>(row) * cols;

    int tid = threadIdx.x;
    int nthreads = blockDim.x;

    // Number of 8-element vectors per row
    int vecCount = cols / 8;
    int tail = cols - vecCount * 8;

    // -----------------------------------------------------------------------
    // Pass 1: compute sum of squares in FP32
    // -----------------------------------------------------------------------
    float localSum = 0.0f;

    // Vectorized portion: each thread processes multiple 8-element vectors
    for (int i = tid; i < vecCount; i += nthreads) {
        // Load 8 BF16 elements as two 64-bit __nv_bfloat1622 values
        // (each __nv_bfloat1622 holds 4 BF16 elements = 64 bits)
        const __nv_bfloat1622* vecPtr = reinterpret_cast<const __nv_bfloat1622*>(rowIn + i * 8);

        // Load first 4 elements (64 bits)
        __nv_bfloat1622 a = vecPtr[0];
        // Load next 4 elements (64 bits)
        __nv_bfloat1622 b = vecPtr[1];

        // Convert to FP32 and accumulate
        float f0 = __bfloat162float(a.x);
        float f1 = __bfloat162float(a.y);
        float f2 = __bfloat162float(a.z);
        float f3 = __bfloat162float(a.w);
        float f4 = __bfloat162float(b.x);
        float f5 = __bfloat162float(b.y);
        float f6 = __bfloat162float(b.z);
        float f7 = __bfloat162float(b.w);

        localSum += f0 * f0 + f1 * f1 + f2 * f2 + f3 * f3;
        localSum += f4 * f4 + f5 * f5 + f6 * f6 + f7 * f7;
    }

    // Scalar tail
    for (int i = vecCount * 8 + tid; i < cols; i += nthreads) {
        float v = __bfloat162float(rowIn[i]);
        localSum += v * v;
    }

    // Block-level reduction of localSum
    // Use shared memory for the reduction
    extern __shared__ float sdata[];
    sdata[tid] = localSum;
    __syncthreads();

    // Tree reduction
    for (int s = nthreads / 2; s > 0; s >>= 1) {
        if (tid < s) {
            sdata[tid] += sdata[tid + s];
        }
        __syncthreads();
    }

    float blockSum = sdata[0];
    float norm = std::sqrt(blockSum);
    float inv = (norm > 0.0f) ? (1.0f / norm) : 0.0f;

    // -----------------------------------------------------------------------
    // Pass 2: divide by norm and store
    // -----------------------------------------------------------------------

    // Vectorized portion
    for (int i = tid; i < vecCount; i += nthreads) {
        const __nv_bfloat1622* vecIn  = reinterpret_cast<const __nv_bfloat1622*>(rowIn + i * 8);
        __nv_bfloat1622*       vecOut = reinterpret_cast<__nv_bfloat1622*>(rowOut + i * 8);

        __nv_bfloat1622 a = vecIn[0];
        __nv_bfloat1622 b = vecIn[1];

        float f0 = __bfloat162float(a.x) * inv;
        float f1 = __bfloat162float(a.y) * inv;
        float f2 = __bfloat162float(a.z) * inv;
        float f3 = __bfloat162float(a.w) * inv;
        float f4 = __bfloat162float(b.x) * inv;
        float f5 = __bfloat162float(b.y) * inv;
        float f6 = __bfloat162float(b.z) * inv;
        float f7 = __bfloat162float(b.w) *
